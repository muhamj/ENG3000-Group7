//  Setup Web Server 
void setupWebServer() {
  // WiFi AP
  WiFi.mode(WIFI_AP);
  WiFi.setSleep(false);
  IPAddress ip(10,10,10,1), gw(10,10,10,1), mask(255,255,255,0);
  WiFi.softAPConfig(ip, gw, mask);
  bool ok = WiFi.softAP(ssid, password, AP_CHANNEL);
  Serial.printf("AP: %s, SSID: %s, IP: %s\n", ok ? "ONLINE" : "FAILED", ssid, WiFi.softAPIP().toString().c_str());

  // Routes
  server.on("/", handleRoot);
  server.on("/health", handleHealth);
  server.on("/emergency", handleEmergency);
  server.on("/reset", handleReset);
  server.on("/overrideOpen", handleOverrideOpen);
  server.on("/overrideClose", handleOverrideClose);
  server.begin();
  Serial.println("Web server started");
}

//  Web Log Buffer 
struct LogBuffer {
  static constexpr size_t CAP = 200;     // number of lines
  String lines[CAP];
  size_t head = 0;                       // next write index
  bool wrapped = false;

  void add(const String& s) {
    lines[head] = s;
    head = (head + 1) % CAP;
    if (head == 0) wrapped = true;
  }

  String dumpHTML() const {
    String out;
    out.reserve(8192);
    out += F("<!doctype html><html><head><meta charset='utf-8'><meta name='viewport' content='width=device-width,initial-scale=1'><title>Logs</title><style>body{font-family:ui-monospace,Menlo,Consolas,monospace;background:#0b0d10;color:#e3e7ee;margin:0}h1{margin:0;padding:12px 16px;background:#171a1f}pre{white-space:pre-wrap;word-wrap:break-word;margin:0;padding:16px}a{color:#9ecbff;text-decoration:none;margin:0 16px}</style></head><body>");
    out += F("<h1>Bridge Logs</h1><div style='padding:8px'><a href='/'>&larr; Back</a><a href='/logs?clear=1' style='float:right'>Clear</a></div><pre>");

    if (wrapped) {
      for (size_t i = head; i < CAP; ++i) { if (lines[i].length()) { out += lines[i]; out += '\n'; } }
    }
    for (size_t i = 0; i < (wrapped ? head : head); ++i) { if (lines[i].length()) { out += lines[i]; out += '\n'; } }

    out += F("</pre></body></html>");
    return out;
  }

  void clear() { head = 0; wrapped = false; for (size_t i=0;i<CAP;++i) lines[i] = String(); }
} gLog;

// ---- UPDATED: Emergency/web-aware wait helper ----
inline bool waitWithEmergency(unsigned long ms){
  unsigned long t0 = millis();
  while (millis() - t0 < ms){
    server.handleClient();
    if (emergencyActive || webPreemptRequested || emergencyPending) return false;  // <-- NEW: preempt
    delay(5);
  }
  return true;
}

void logicSetup() {
  Serial.begin(115200);
  delay(200);

  // NeoPixels
  strip1.begin();
  strip2.begin();
  strip1.show(); 
  strip2.show();

  // Motor pins
  pinMode(EN1, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);
  stopMotor();

  // Ultrasonic pins
  pinMode(LEFT_TRIG, OUTPUT);
  pinMode(LEFT_ECHO, INPUT);
  pinMode(RIGHT_TRIG, OUTPUT);
  pinMode(RIGHT_ECHO, INPUT);

  // Initial state
  setAll(strip2, strip2.Color(0, 255, 0)); // Cars green
  setAll(strip1, strip1.Color(255, 0, 0)); // Boats red
  strip1.show();
  strip2.show();
  
  // Limit Switches (INPUT_PULLUP -> pressed == LOW)
  pinMode(limitSwitchOpen,  INPUT_PULLUP);
  pinMode(limitSwitchClose, INPUT_PULLUP);

  // Emergency button
  pinMode(EMERGENCY_PIN, INPUT_PULLUP);
  attachInterrupt(digitalPinToInterrupt(EMERGENCY_PIN), emergencyISR, FALLING);

  // Override buttons up and down
  pinMode(OVERRIDE_OPEN_PIN, INPUT_PULLUP);
  pinMode(OVERRIDE_CLOSE_PIN, INPUT_PULLUP);

  // Buzzer
  pinMode(buzzerPin, OUTPUT);
}

void logicLoop() {
  if (emergencyPending) { 
    emergencyPending = false; 
    enterEmergency();
    return;
    }

  // ---- Immediate service of WEB overrides (pre-clears preempt so we can run) ----
  if (webOverrideOpenRequest) {
    webOverrideOpenRequest = false;
    webPreemptRequested = false;          // <-- consume preempt
    Serial.println("Web Override: OPEN bridge");
    emergencyActive = false;
    stopMotor();
    setAll(strip2, strip2.Color(255,0,0)); strip2.show();
    setAll(strip1, strip1.Color(255,0,0)); strip1.show();
    if (!waitWithEmergency(20)) return;

    bool hitOpen = driveUntilLimit(backward, 255, limitSwitchOpen, 15000);
    if (!hitOpen) Serial.println("WARNING: Open limit not reached.");

    setAll(strip1, strip1.Color(0,255,0)); strip1.show();
    setAll(strip2, strip2.Color(255,0,0)); strip2.show();
    bridgeOpen = true;
    motorDirection = false;
    bridgeOpenTime = millis();
  }

  if (webOverrideCloseRequest) {
    webOverrideCloseRequest = false;
    webPreemptRequested = false;          // <-- consume preempt
    Serial.println("Web Override: CLOSE bridge");
    emergencyActive = false;
    stopMotor();
    setAll(strip1, strip1.Color(255,0,0)); strip1.show();
    delay(300);

    // ===== CHANGED: choose direction based on motorDirection =====
    bool hitClose = false;
    if (motorDirection == false) {
      // last open was backward -> close forward
      hitClose = driveUntilLimit(forward, 255, limitSwitchClose, 15000);
    } else {
      // last open was forward -> close backward
      hitClose = driveUntilLimit(backward, 255, limitSwitchClose, 15000);
    }
    // ===== END CHANGE =====

    if (!hitClose) Serial.println("WARNING: Close limit not reached.");

    setAll(strip2, strip2.Color(0,255,0)); strip2.show();
    setAll(strip1, strip1.Color(255,0,0)); strip1.show();
    bridgeOpen = false;
  }

  // Emergency handling first
  if (emergencyActive) {
    stopMotor();

    // Siren effect
    if (millis() - lastSirenTime >= 10) {
      lastSirenTime = millis();
      if (sirenDirection) {
        sirenFreq += 20;
        if (sirenFreq >= 1200) sirenDirection = false;
      } else {
        sirenFreq -= 20;
        if (sirenFreq <= 800) sirenDirection = true;
      }
      tone(buzzerPin, sirenFreq);
    }
    
    // Flash all lights red 
    if (millis() - lastFlashTime >= 500) {
      flashState = !flashState;
      lastFlashTime = millis();
      if (flashState) {
        setAll(strip2, strip2.Color(255, 0, 0)); strip2.show();
        setAll(strip1, strip1.Color(255, 0, 0)); strip1.show(); 
      } else {
        setAll(strip1, strip1.Color(0, 0, 0));
        setAll(strip2, strip2.Color(0, 0, 0));
      }
      strip1.show();
      strip2.show();
    }
    Serial.println("EMERGENCY STOP ACTIVE");
    delay(100); 

    //  EMERGENCY OVERRIDES (leave emergency into OPEN/CLOSED) 

    //  Emergency -> OPEN 
    if (digitalRead(OVERRIDE_OPEN_PIN) == LOW) {
      Serial.println("Manual Override (EMERGENCY): OPEN bridge");
      stopMotor();
      noTone(buzzerPin);                 // stop siren
      // Briefly hold both red during transition
      setAll(strip2, strip2.Color(255, 0, 0)); strip2.show();  // Cars red
      setAll(strip1, strip1.Color(255, 0, 0)); strip1.show();  // Boats red
      delay(300);

      // Allow motion by clearing emergency before driving
      emergencyActive = false;

      // Open bridge; original used backward()
      bool hitOpen = driveUntilLimit(backward, 255, limitSwitchOpen, 15000);
      if (!hitOpen) Serial.println("WARNING: Open limit not reached.");

      // Enter normal OPEN state
      setAll(strip1, strip1.Color(0, 255, 0)); strip1.show();  // Boats green
      setAll(strip2, strip2.Color(255, 0, 0)); strip2.show();  // Cars red
      bridgeOpen      = true;
      motorDirection  = false;  // last open used backward()
      bridgeOpenTime  = millis();

      // Debounce button release
      while (digitalRead(OVERRIDE_OPEN_PIN) == LOW) delay(10);
      return; // skip rest of loop this cycle
    }

    //  Emergency -> CLOSE 
    if (digitalRead(OVERRIDE_CLOSE_PIN) == LOW) {
      Serial.println("Manual Override (EMERGENCY): CLOSE bridge");
      stopMotor();
      noTone(buzzerPin);                 // stop siren
      setAll(strip1, strip1.Color(255, 0, 0)); strip1.show();  // Boats red
      delay(300);

      // Allow motion by clearing emergency before driving
      emergencyActive = false;

      // Close bridge (original emergency close used forward)
      bool hitClose = driveUntilLimit(forward, 255, limitSwitchClose, 15000);
      if (!hitClose) Serial.println("WARNING: Close limit not reached.");

      // Enter normal CLOSED state
      setAll(strip2, strip2.Color(0, 255, 0)); strip2.show();  // Cars green
      setAll(strip1, strip1.Color(255, 0, 0)); strip1.show();  // Boats red
      bridgeOpen = false;

      while (digitalRead(OVERRIDE_CLOSE_PIN) == LOW) delay(10);
      return; // skip rest of loop this cycle
    }

    // No override pressed: stay in emergency behavior
    return; 
  } else {
    noTone(buzzerPin); // Turn off siren when not in emergency
  }

  // Read sensors with crosstalk guard
  long farcm   = readDistanceCM(LEFT_TRIG, LEFT_ECHO);    // FAR sensor
  delay(50); // allow echoes to dissipate before next ping
  long closecm = readDistanceCM(RIGHT_TRIG, RIGHT_ECHO);  // CLOSE sensor

  Serial.print("Far sensor: ");
  Serial.print(farcm);
  Serial.print("cm, Close sensor: ");
  Serial.print(closecm);
  Serial.print("cm, Bridge: ");
  Serial.println(bridgeOpen ? "OPEN" : "CLOSED");

  //  OPENING SEQUENCE (sensor facing me / FAR triggers) 
  if (farcm > 0 && farcm <= THRESHOLD_CM && !bridgeOpen) {
    Serial.println("Sensor Triggered (FAR) - Opening, Cars stopping.");

    // Traffic lights sequencing
    setAll(strip2, strip2.Color(255, 165, 0)); strip2.show(); // cars = orange
    if (!waitWithEmergency(4000)) return;
    setAll(strip2, strip2.Color(255, 0, 0)); strip2.show();   // cars = red
    if (!waitWithEmergency(4000)) return;
    setAll(strip1, strip1.Color(255, 0, 0)); strip1.show();   // boats = red
    if (!waitWithEmergency(4000)) return;

    // Open bridge: original code used backward() here
    bool hitOpen = driveUntilLimit(backward, 255, limitSwitchOpen, 15000);
    if (!hitOpen) {
      Serial.println("WARNING: Open limit not reached (timeout or emergency).");
    }

    // Boats green
    if (!waitWithEmergency(2500)) return;
    setAll(strip1, strip1.Color(0, 255, 0)); strip1.show();
    Serial.println("Bridge State: Open. Traffic: Boats passing.");

    bridgeOpen = true;
    motorDirection = false; // last open used backward()
    bridgeOpenTime = millis();
  }

  //  OPENING SEQUENCE (sensor facing away / CLOSE triggers) 
  if (closecm > 0 && closecm <= THRESHOLD_CM && !bridgeOpen) {
    Serial.println("Sensor Triggered (CLOSE) - Opening, Cars stopping.");

    // Traffic lights sequencing
    setAll(strip2, strip2.Color(255, 165, 0)); strip2.show(); // cars = orange
    if (!waitWithEmergency(4000)) return;
    setAll(strip2, strip2.Color(255, 0, 0)); strip2.show();   // cars = red
    if (!waitWithEmergency(4000)) return;
    setAll(strip1, strip1.Color(255, 0, 0)); strip1.show();   // boats = red
    if (!waitWithEmergency(4000)) return;

    // Open bridge: original code used forward() here
    bool hitOpen2 = driveUntilLimit(forward, 255, limitSwitchOpen, 15000);
    if (!hitOpen2) {
      Serial.println("WARNING: Open limit not reached (timeout or emergency).");
    }

    // Boats green
    if (!waitWithEmergency(2500)) return;
    setAll(strip1, strip1.Color(0, 255, 0)); strip1.show();
    Serial.println("Bridge State: Open. Traffic: Boats passing.");

    bridgeOpen = true;
    motorDirection = true; // last open used forward()
    bridgeOpenTime = millis();
  }

  //  CLOSING SEQUENCE 
  // Bridge is open AND no boat detected by either sensor AND minimum open time passed
  if (bridgeOpen && (farcm > THRESHOLD_CM || farcm <= 0) && (closecm > THRESHOLD_CM || closecm <= 0)) {
    if (millis() - bridgeOpenTime >= MIN_OPEN_TIME) {
      Serial.println("No boat detected, closing sequence starting");

      // Boats to red 
      if (!waitWithEmergency(5000)) return;
      setAll(strip1, strip1.Color(255, 0, 0)); strip1.show(); 
      Serial.println("Boats: RED");
      if (!waitWithEmergency(5000)) return;

      Serial.println("Closing bridge");
      bool hitClose = false;
      if (motorDirection == false) {
        // last open was backward -> now close forward
        hitClose = driveUntilLimit(forward, 255, limitSwitchClose, 15000);
      } else {
        // last open was forward -> now close backward
        hitClose = driveUntilLimit(backward, 255, limitSwitchClose, 15000);
      }
      if (!hitClose) {
        Serial.println("WARNING: Close limit not reached (timeout or emergency).");
      }

      Serial.println("Bridge State: Closed. Traffic: Cars resuming.");
      
      // Cars back to green, boats stay red
      if (!waitWithEmergency(5000)) return;
      setAll(strip2, strip2.Color(0, 255, 0)); strip2.show(); // cars = green
      setAll(strip1, strip1.Color(255, 0, 0)); strip1.show(); // boats = red
      if (!waitWithEmergency(5000)) return;
      
      // Bridge is now closed
      bridgeOpen = false;
      Serial.println("Bridge fully closed");
    }
  }

  // Keep bridge open if a boat is still detected
  if (bridgeOpen && ((farcm > 0 && farcm <= THRESHOLD_CM) || (closecm > 0 && closecm <= THRESHOLD_CM))) {
    setAll(strip1, strip1.Color(0, 255, 0)); strip1.show(); // boats green
    setAll(strip2, strip2.Color(255, 0, 0)); strip2.show(); // cars red
    Serial.println("Boat still detected");
    bridgeOpenTime = millis();
  }

  if (!waitWithEmergency(500)) return;
}

//  Helpers 
long readDistanceCM(int trigPin, int echoPin) {
  // Ensure ECHO is low before starting a new measurement
  unsigned long guardStart = millis();
  while (digitalRead(echoPin) == HIGH && millis() - guardStart < 5) {
    delayMicroseconds(50);
  }

  digitalWrite(trigPin, LOW);
  delayMicroseconds(5);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  // Increased timeout for up to (and beyond) 5 m
  long duration = pulseIn(echoPin, HIGH, 60000UL); // 60ms ≈ ~10m headroom
  if (duration == 0) return -1; // no reading
  long cm = duration / 29 / 2;
  return cm;
}

// Motor control

void backward(int speed) {
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);

void stopMotor() {
  // Coast stop
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
  analogWrite(EN1, 0);
}

void enterEmergency() {
  emergencyActive = true;
  stopMotor();
  webPreemptRequested = true;

  sirenFreq = 800;
  sirenDirection = true;
  setAll(strip1, strip1.Color(255, 0, 0));  // Boats red
  setAll(strip2, strip2.Color(255, 0, 0));  // Cars red
  strip1.show();
  strip2.show();
}


// Debounced read for INPUT_PULLUP limit switch (pressed == LOW)
bool limitPressed(int pin, unsigned long stableMs) {
  if (digitalRead(pin) == LOW) {
    unsigned long t0 = millis();
    while (millis() - t0 < stableMs) {
      if (digitalRead(pin) != LOW) return false;
      delay(1);
    }
    return true;
  }
  return false;
}

// ---- UPDATED: Drive until limit pressed / emergency / preempt / timeout ----
bool driveUntilLimit(void (*driveFn)(int), int speed, int limitPin, unsigned long timeoutMs) {
  unsigned long t0 = millis();
  driveFn(speed);  // start moving

  while (millis() - t0 < timeoutMs) {
    // Service web requests so override buttons are responsive during motion
    server.handleClient();
    if (emergencyActive || webPreemptRequested || emergencyPending) {     // <-- NEW: preempt
      stopMotor();
      Serial.println("Preempted (emergency/web) -> motor stopped");
      return false;
    }
    if (limitPressed(limitPin, 25)) { // pressed & stable
      stopMotor();
      Serial.println("Limit reached -> motor stopped");
      return true;
    }
    delay(5); // small yield
  }

  stopMotor(); // safety
  Serial.println("Timeout waiting for limit -> motor stopped");
  return false;
}

// ====== UPDATED WEB HANDLERS (only act during EMERGENCY; exit it and perform sequence) ======
void handleReset() {
  // Only allow reset while in EMERGENCY
  if (!emergencyActive) {
    Serial.println("Reset ignored (not in EMERGENCY).");
    server.sendHeader("Location", "/");
    server.send(302, "text/plain", "");
    return;
  }

  Serial.println("RESET: Restoring default state from EMERGENCY.");
  emergencyActive = false;       // exit emergency
  noTone(buzzerPin);
  stopMotor();

  // Default state: bridge CLOSED, Cars GREEN, Boats RED
  bridgeOpen = false;
  setAll(strip1, strip1.Color(255, 0, 0)); // boats red
  setAll(strip2, strip2.Color(0, 255, 0)); // cars green
  strip1.show();
  strip2.show();

  webPreemptRequested = true;    // break out of any waits/motion immediately
  server.sendHeader("Location", "/");
  server.send(302, "text/plain", "");
}

void handleBridgeFlags() {

    // anchor: start of handleBridgeFlags
  if (emergencyPending) { 
    emergencyPending = false; 
    enterEmergency(); 
    // continue into the emergencyActive block below this return path
  }

  //  EMERGENCY 
  if (emergencyActive) {
    stopMotor();

    // Siren
    if (millis() - lastSirenTime >= 10) {
      lastSirenTime = millis();
      if (sirenDirection) {
        sirenFreq += 20;
        if (sirenFreq >= 1200) sirenDirection = false;
      } else {
        sirenFreq -= 20;
        if (sirenFreq <= 800) sirenDirection = true;
      }
      tone(buzzerPin, sirenFreq);
    }

    // Flash lights red
    if (millis() - lastFlashTime >= 500) {
      flashState = !flashState;
      lastFlashTime = millis();
      if (flashState) {
        setAll(strip2, strip2.Color(255,0,0)); strip2.show();
        setAll(strip1, strip1.Color(255,0,0)); strip1.show();
      } else {
        setAll(strip1, strip1.Color(0,0,0));
        setAll(strip2, strip2.Color(0,0,0));
      }
    }

    Serial.println("EMERGENCY STOP ACTIVE");

    // --- Web overrides honored during EMERGENCY (mirror manual behavior) ---
    if (webOverrideOpenRequest) {
      Serial.println("Web Override (EMERGENCY): OPEN bridge");
      webOverrideOpenRequest = false;
      stopMotor();
      noTone(buzzerPin);
      // show both red briefly for UI consistency
      setAll(strip2, strip2.Color(255, 0, 0)); strip2.show();
      setAll(strip1, strip1.Color(255, 0, 0)); strip1.show();
      delay(300);

      emergencyActive = false;           // leave emergency before motion
      webPreemptRequested = false;       // <<< FIX: consume preempt so waits resume normally

      bool hitOpen = driveUntilLimit(backward, 255, limitSwitchOpen, 15000);
      if (!hitOpen) Serial.println("WARNING: Open limit not reached.");

      setAll(strip1, strip1.Color(0, 255, 0)); strip1.show();  // Boats green
      setAll(strip2, strip2.Color(255, 0, 0)); strip2.show();  // Cars red
      bridgeOpen      = true;
      motorDirection  = false;
      bridgeOpenTime  = millis();
      return;
    }

    if (webOverrideCloseRequest) {
      Serial.println("Web Override (EMERGENCY): CLOSE bridge");
      webOverrideCloseRequest = false;
      stopMotor();
      noTone(buzzerPin);
      setAll(strip1, strip1.Color(255, 0, 0)); strip1.show();  // Boats red
      delay(300);

      emergencyActive = false;           // leave emergency before motion
      webPreemptRequested = false;       // <<< FIX: consume preempt

      // ===== CHANGED: choose direction based on motorDirection =====
      bool hitClose = false;
      if (motorDirection == false) {
        // last open was backward -> close forward
        hitClose = driveUntilLimit(forward, 255, limitSwitchClose, 15000);
      } else {
        // last open was forward -> close backward
        hitClose = driveUntilLimit(backward, 255, limitSwitchClose, 15000);
      }
      // ===== END CHANGE =====

      if (!hitClose) Serial.println("WARNING: Close limit not reached.");

      setAll(strip2, strip2.Color(0, 255, 0)); strip2.show();  // Cars green
      setAll(strip1, strip1.Color(255, 0, 0)); strip1.show();  // Boats red
      bridgeOpen = false;
      return;
    }
    // -----------------------------------------------------------------------

    // Manual override buttons during emergency still work
    if (digitalRead(OVERRIDE_OPEN_PIN) == LOW)  { bridgeOpenRequest = true; emergencyActive = false; }
    if (digitalRead(OVERRIDE_CLOSE_PIN) == LOW) { bridgeCloseRequest = true; emergencyActive = false; }

    delay(100); // short dwell
    return; // skip other logic while emergency
  }

  noTone(buzzerPin); // turn off siren if not emergency

  // ----- BRIDGE OPEN REQUEST -----
  if (webOverrideOpenRequest || bridgeOpenRequest) {
    webOverrideOpenRequest = false;
    bridgeOpenRequest = false;
    webPreemptRequested = false;   // <-- consume preempt

    Serial.println("Bridge OPEN request received");
    stopMotor();
    setAll(strip2, strip2.Color(255,0,0)); strip2.show();
    setAll(strip1, strip1.Color(255,0,0)); strip1.show();
    if (!waitWithEmergency(300)) return;

    bool hitOpen = driveUntilLimit(backward, 255, limitSwitchOpen, 15000);
    if (!hitOpen) Serial.println("WARNING: Open limit not reached.");

    setAll(strip1, strip1.Color(0,255,0)); strip1.show();
    setAll(strip2, strip2.Color(255,0,0)); strip2.show();
    bridgeOpen = true;
    motorDirection = false;
    bridgeOpenTime = millis();
  }

  //  BRIDGE CLOSE REQUEST 
  if (webOverrideCloseRequest || bridgeCloseRequest) {
    webOverrideCloseRequest = false;
    bridgeCloseRequest = false;
    webPreemptRequested = false;   // <-- consume preempt

    Serial.println("Bridge CLOSE request received");
    stopMotor();
    setAll(strip1, strip1.Color(255,0,0)); strip1.show();
    delay(300);

    // ===== CHANGED: choose direction based on motorDirection =====
    bool hitClose = false;
    if (motorDirection == false) {
      // last open was backward -> close forward
      hitClose = driveUntilLimit(forward, 255, limitSwitchClose, 15000);
    } else {
      // last open was forward -> close backward
      hitClose = driveUntilLimit(backward, 255, limitSwitchClose, 15000);
    }
    // ===== END CHANGE =====

    if (!hitClose) Serial.println("WARNING: Close limit not reached.");

    setAll(strip2, strip2.Color(0,255,0)); strip2.show();
    setAll(strip1, strip1.Color(255,0,0)); strip1.show();
    bridgeOpen = false;
  }
}

  Serial.begin(115200);

void loop() {
  // Service web requests continuously for best responsiveness
  webLoop();
  if (emergencyPending) { emergencyPending = false; enterEmergency(); }  // NEW
  handleBridgeFlags();
  if (emergencyActive) return;
  logicLoop();      // run the untouched bridge logic (with patched emergency block)
  // webLoop() also called frequently inside waits/motion
}
